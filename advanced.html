01 · Modern C++

先搞清楚"C++ 有好几个版本"

网上很多教程讲的是 C++98 的写法，而实际项目早就用 C++17 / C++20 了。 先知道自己学的是哪一版，能省下大量困惑。 


#include <iostream>
#include <vector>
#include <map>
#include <string>
#include <optional>
#include <variant>
#include <string_view>
#include <algorithm>
#include <numeric>

// C++17：结构化绑定让遍历 map 清爽很多
void printScores(const std::map<std::string, int>& scores) {
    for (const auto& [name, score] : scores)          // → 结构化绑定
        std::cout << name << " = " << score << '\n';
}

// C++14：泛型 lambda + auto 返回类型推导
auto sumOf = [](const auto& container) {
    return std::accumulate(container.begin(), container.end(), 0);
};

// C++17：optional 表达"可能没有值"，替代返回 -1 或空指针
std::optional<int> findIndex(const std::vector<int>& v, int target) {
    auto it = std::find(v.begin(), v.end(), target);
    if (it == v.end()) return std::nullopt;
    return (int)std::distance(v.begin(), it);
}

// C++17：variant 表达"几个类型中的一个"，替代 union + 手动记 tag
using Value = std::variant<int, double, std::string>;

std::string describe(const Value& v) {
    return std::visit([](const auto& x) -> std::string {
        using T = std::decay_t<decltype(x)>;
        if constexpr (std::is_same_v<T, int>)         return "整数: " + std::to_string(x);
        else if constexpr (std::is_same_v<T, double>) return "小数: " + std::to_string(x);
        else                                          return "字符串: " + x;
    }, v);
}

// C++17：string_view 传字符串不拷贝（只读视图）
std::size_t countChar(std::string_view sv, char c) {
    return (std::size_t)std::count(sv.begin(), sv.end(), c);
}

// C++11 起：constexpr 让计算发生在编译期
constexpr long long factorial(int n) {
    return n <= 1 ? 1 : n * factorial(n - 1);
}
static_assert(factorial(5) == 120, "编译期就能验证！");

int main() {
    printScores({{"小明", 92}, {"小红", 98}, {"小 C", 100}});

    std::vector<int> v = {10, 20, 30, 40};
    std::cout << sumOf(v) << '\n';                    // 100

    if (auto idx = findIndex(v, 30))                    // C++17：if 里带初始化
        std::cout << "找到，下标 = " << *idx << '\n';   // 2
    else
        std::cout << "没找到\n";

    std::cout << describe(Value{42}) << '\n';
    std::cout << describe(Value{3.14}) << '\n';
    std::cout << describe(Value{std::string("hello")}) << '\n';

    std::cout << countChar("banana", 'a') << '\n';      // 3
    std::cout << "5! = " << factorial(5) << '\n';       // 120
    return 0;
}
02 · RAII & Smart Pointers

RAII：C++ 最核心的一条设计哲学

把资源的生命周期绑在对象的生命周期上。 在构造函数里获取资源，在析构函数里释放资源。只要对象被正确销毁，资源就一定会被释放—— 哪怕中途抛出异常。

unique_ptr 独占

一个对象只能有一个所有者、不能拷贝，只能 move。 零额外开销，应该作为默认首选。

shared_ptr 共享

通过引用计数共享所有权，计数归零才释放。 有原子操作开销（不要默认用它）。

weak_ptr 观察

不增加引用计数，用来打破循环引用造成的泄漏。 使用前要用 lock() 检查对象是否还活着。 


#include <iostream>
#include <memory>
#include <vector>
#include <string>

struct Resource {
    std::string name;
    explicit Resource(std::string n) : name(std::move(n)) {
        std::cout << "  获取资源: " << name << '\n';
    }
    ~Resource() {
        std::cout << "  释放资源: " << name << '\n';   // 一定会被调用
    }
};

// ---------- RAII 手写版：任何资源都能这样封装 ----------
class FileGuard {
    std::FILE* fp_ = nullptr;
public:
    explicit FileGuard(const char* path, const char* mode) : fp_(std::fopen(path, mode)) {
        if (!fp_) throw std::runtime_error("打开文件失败");
    }
    ~FileGuard() { if (fp_) std::fclose(fp_); }        // 析构时自动关问

    FileGuard(const FileGuard&)            = delete;    // 禁止拷贝
    FileGuard& operator=(const FileGuard&) = delete;
    FileGuard(FileGuard&& o) noexcept : fp_(o.fp_) { o.fp_ = nullptr; }  // 允许移动

    std::FILE* get() const { return fp_; }
};

// ---------- shared_ptr 的经典坑：循环引用 ----------
struct NodeBad {
    std::shared_ptr<NodeBad> next;
    ~NodeBad() { std::cout << "  NodeBad 析构\n"; }
};

struct NodeGood {
    std::shared_ptr<NodeGood> next;
    std::weak_ptr<NodeGood>   prev;      // ★ 反向用 weak_ptr，打破循环
    ~NodeGood() { std::cout << "  NodeGood 析构\n"; }
};

int main() {
    std::cout << "[1] 作用域结束自动释放\n";
    {
        auto r = std::make_unique<Resource>("独占地雷");
    }   // 出作用域立刻打印"释放资源"

    std::cout << "[2] 所有权转移\n";
    {
        auto p = std::make_unique<Resource>("可转移的宝物");
        auto q = std::move(p);                      // p 变成 nullptr
        std::cout << "  p 现在为空? " << std::boolalpha << (p == nullptr) << '\n';
    }

    std::cout << "[3] 循环引用导致泄漏（两个都不会析构）\n";
    {
        auto a = std::make_shared<NodeBad>();
        auto b = std::make_shared<NodeBad>();
        a->next = b;
        b->next = a;                                // 计数永远归不了 0
        std::cout << "  a.use_count = " << a.use_count() << '\n';
    }

    std::cout << "[4] 用 weak_ptr 打破循环（两个都会正常析构）\n";
    {
        auto a = std::make_shared<NodeGood>();
        auto b = std::make_shared<NodeGood>();
        a->next = b;
        b->prev = a;                                // weak_ptr 不加计数
        std::cout << "  a.use_count = " << a.use_count() << '\n';
    }
    return 0;
}
🧠
为什么优先用 make_unique / make_shared
  • 异常安全：f(std::shared_ptr<T>(new T), g()) 中如果 g() 抛异常，new T 的内存会泄漏（make_shared 不会）
  • 更高效：make_shared 一次性分配对象和控制块，少一次内存分配。
  • 更简洁：不用重复写类型名。
  • 注意：make_shared 会让对象内存和控制块同生共死，如果有 weak_ptr 长期存在，对象内存会迟迟无法归还。
03 · Move Semantics

移动语义：把"拷贝"变成"转移"

拷贝一个装了一百万个元素的 vector 很贵；但如果你只是想要它的数据、不再需要原来那个，那就没必要复制——直接把内部的指针交出去就行。这就是移动。 


#include <iostream>
#include <vector>
#include <string>
#include <utility>
#include <chrono>

class Buffer {
    char*  data_ = nullptr;
    size_t size_ = 0;
public:
    explicit Buffer(size_t n) : data_(new char[n]), size_(n) {
        std::cout << "  构造 " << size_ << " 字节\n";
    }

    // ---- 析构 ----
    ~Buffer() { delete[] data_; }

    // ---- 拷贝构造：深拷贝，O(n) ----
    Buffer(const Buffer& o) : data_(new char[o.size_]), size_(o.size_) {
        std::copy(o.data_, o.data_ + o.size_, data_);
        std::cout << "  拷贝构造（慢，O(n)）\n";
    }

    // ---- 移动构造：只偷指针，O(1) ----
    Buffer(Buffer&& o) noexcept
        : data_(o.data_), size_(o.size_) {
        o.data_ = nullptr;               // ★ 必须让原对象放弃所有权，否则会 double free
        o.size_ = 0;
        std::cout << "  移动构造（快，O(1)）\n";
    }

    // ---- 拷贝赋值：copy-and-swap，天然异常安全 ----
    Buffer& operator=(const Buffer& o) {
        if (this != &o) { Buffer tmp(o); swap(tmp); }
        return *this;
    }

    // ---- 移动赋值 ----
    Buffer& operator=(Buffer&& o) noexcept {
        if (this != &o) {
            delete[] data_;                          // 先释放自己的
            data_ = o.data_; size_ = o.size_;
            o.data_ = nullptr; o.size_ = 0;
        }
        return *this;
    }

    void swap(Buffer& o) noexcept {
        std::swap(data_, o.data_);
        std::swap(size_, o.size_);
    }
    size_t size() const { return size_; }
};

int main() {
    std::cout << "[拷贝] 代价与数据量成正比\n";
    {
        Buffer a(1'000'000);
        Buffer b = a;                    // 调用拷贝构造
    }

    std::cout << "[移动] 代价是常数\n";
    {
        Buffer a(1'000'000);
        Buffer b = std::move(a);         // 调用移动构造
    }

    std::cout << "[返回大对象：编译器会做 RVO / NRVO，连移动都省了]\n";
    auto makeBig = [] {
        std::vector<int> v(1'000'000, 7);
        return v;                        // 不会拷贝，直接在调用方位置构造
    };
    auto v = makeBig();

    std::cout << "[emplace_back vs push_back]\n";
    std::vector<std::string> vs;
    vs.reserve(2);
    vs.push_back("临时字符串");           // 先构造临时对象，再移动进来
    vs.emplace_back(10, 'x');            // 直接在容器内存里构造，少一次移动
    std::cout << "  vs[1] = " << vs[1] << '\n';

    std::cout << "[完美转发：forward 保持参数的左右值属性]\n";
    auto wrapper = [](auto&& x) {
        return std::string(std::forward<decltype(x)>(x));
    };
    std::string s = "hello";
    std::cout << "  " << wrapper(s) << " / " << wrapper(std::string("world")) << '\n';
    return 0;
}
⚠️
三个最常见的移动语义错误
  1. 移动构造函数忘了把源对象置空 → 两个对象指向同一块内存，析构时 double free 崩溃。
  2. 对 const 对象使用 std::move → const T&& 匹配不到移动构造，静默退化成拷贝，你以为优化了其实没有。
  3. 移动之后继续使用源对象 → 状态未指定。唯一的合法操作是赋值或销毁。
04 · Concurrency

并发：从"能跑"到"跑对"

多线程最大的特点不是"难写"，而是错误会随机出现、难以复现。 所以必须从原理上理解数据竞争，而不是靠试。 


#include <iostream>
#include <thread>
#include <mutex>
#include <atomic>
#include <condition_variable>
#include <queue>
#include <vector>
#include <chrono>

// ---------- 1. 基础：启动线程并等待 ----------
void basicThread() {
    std::vector<std::thread> pool;
    for (int i = 0; i < 4; ++i)
        pool.emplace_back([i] { std::cout << "  线程 " << i << " 工作中\n"; });
    for (auto& t : pool) t.join();          // ★ 必须 join 或 detach，否则 terminate
}

// ---------- 2. 数据竞争：错误的写法 ----------
long long counterWrong = 0;
void addWrong(int n) { for (int i = 0; i < n; ++i) ++counterWrong; }

// ---------- 3. 用 mutex 修正 ----------
long long counterMutex = 0;
std::mutex mtx;
void addMutex(int n) {
    for (int i = 0; i < n; ++i) {
        std::lock_guard<std::mutex> lock(mtx);   // 出作用域自动解锁（RAII）
        ++counterMutex;
    }
}

// ---------- 4. 用 atomic 修正（更轻量） ---------
std::atomic<long long> counterAtomic{0};
void addAtomic(int n) { for (int i = 0; i < n; ++i) ++counterAtomic; }

// ---------- 5. 生产者—消费者：条件变量 ----------
template <typename T>
class BlockingQueue {
    std::queue<T> q_;
    std::mutex m_;
    std::condition_variable cv_;
    bool closed_ = false;
public:
    void push(T value) {
        {
            std::lock_guard<std::mutex> lock(m_);
            q_.push(std::move(value));
        }
        cv_.notify_one();                    // ★ 先解锁再通知，减少无谓等待
    }

    bool pop(T& out) {
        std::unique_lock<std::mutex> lock(m_);
        cv_.wait(lock, [this] { return !q_.empty() || closed_; });  // 谓词版，防虚假唤醒
        if (q_.empty()) return false;
        out = std::move(q_.front());
        q_.pop();
        return true;
    }

    void close() {
        { std::lock_guard<std::mutex> lock(m_); closed_ = true; }
        cv_.notify_all();
    }
};

int main() {
    const int N = 200'000;

    std::cout << "[1] 基础线程\n";
    basicThread();

    std::cout << "[2] 无保护（结果几乎肯定不对）\n";
    {
        std::thread a(addWrong, N), b(addWrong, N);
        a.join(); b.join();
        std::cout << "  期望 " << 2 * N << "，实际 " << counterWrong
                  << (counterWrong == 2LL * N ? "" : "   → 数据竞争！") << '\n';
    }

    std::cout << "[3] mutex 保护\n";
    {
        std::thread a(addMutex, N), b(addMutex, N);
        a.join(); b.join();
        std::cout << "  结果 " << counterMutex << '\n';
    }

    std::cout << "[4] atomic\n";
    {
        std::thread a(addAtomic, N), b(addAtomic, N);
        a.join(); b.join();
        std::cout << "  结果 " << counterAtomic.load() << '\n';
    }

    std::cout << "[5] 生产者—消费者\n";
    {
        BlockingQueue<int> bq;
        std::thread producer([&] {
            for (int i = 1; i <= 5; ++i) { bq.push(i * 10); }
            bq.close();
        });
        std::thread consumer([&] {
            int v;
            while (bq.pop(v)) std::cout << "  消费到 " << v << '\n';
        });
        producer.join();
        consumer.join();
    }
    return 0;
}
✓
并发安全清单
  • 每个 thread 都必须 join 成 detach
  • 优先用 lock_guard，需要手动解锁才用 unique_lock
  • 同时锁多个 mutex 用 std::scoped_lock，避免死错
  • condition_variable 一定要用带谓词的 wait
  • 临界区尽可能短，别在锁里做 IO 或耗时计算
🚨
常见误解
  • "加个 volatile 就线程安全了" —— 错，volatile 不提供原子性和内存序
  • "两个线程读同一个变量也危险" —— 只读且没人写是安全的
  • "原子变量就能解决一切" —— 多个原子变量之间仍需注意顺序
  • "我的机器上跑对了很多次" —— 不代表没有竞争
05 · Build System

CMake：让代码在任何机器上都能构建

手敲 g++ a.cpp b.cpp -o app 只适合单文件练习。 只要有 3 个以上源文件、或者要用第三方库，就该上 CMake 了。 


# ================ CMakeLists.txt（放在项目根目录） ===============
cmake_minimum_required(VERSION 3.20)

project(CodeBloom
    VERSION 1.0.0
    DESCRIPTION "C++ 学习示例项目"
    LANGUAGES CXX)

# ---- 全局 C++ 标准 ----
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)          # 用 -std=c++17 而不是 -std=gnu++17

# ---- 默认构建类型（多配置生成器如 VS 会忽略这一行）----
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "" FORCE)
endif()

# ---- 把警告当回事：新项目建议一路开到 -Werror ----
if(MSVC)
    add_compile_options(/W4 /permissive- /Zc:__cplusplus)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()

# ================ 一个库：把逻辑代码收进去 ================
# 现代 CMake 的核心思想：一切都是 target（目标），依赖挂在 target 上
add_library(core STATIC
    src/math_utils.cpp
    src/string_utils.cpp)

# 头文件目录只对自己和链接它的人生效（PUBLIC）
target_include_directories(core PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/include)

# ================ 一个可执行文件 ================
add_executable(app src/main.cpp)
target_link_libraries(app PRIVATE core)

# ================ 单元测试 ================
enable_testing()

# 方式一：用 FetchContent 自动拉取 GoogleTest（首次配置需要联网）
include(FetchContent)
FetchContent_Declare(
    googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG        v1.14.0)

# Windows 下让 gtest 用动态运行时，避免 CRT 冲突
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

add_executable(unit_tests tests/test_math_utils.cpp)
target_link_libraries(unit_tests PRIVATE core GTest::gtest_main)

include(GoogleTest)
gtest_discover_tests(unit_tests)

06 · Test & Debug

测试与调试：把 bug 挡在提交之前

一个真实规律：写测试花的时间，永远少于手工点按钮测出 bug 再回头定位的时间。 尤其是 C++ —— 内存错误往往在崩的那一刻早已远离了出错的那一刻。

}

三个轻量替代方案

  • doctest —— 单头文件，编译极快，语法几乎和 GoogleTest 一样。适合小项目。
  • Catch2 —— 单头文件（v2），断言可以写成自然语言风格。
  • MSVC 自带的 C++ 测试框架 —— 在 Visual Studio 里新建项直接选"本机单元测试"，零配置。

测什么

  • 边界值， 。 、空、最大值、最小值
  • 特殊情况：全相同、已排序、逆序、负数
  • 异常路径：非法输入是否正确抛出
  • 不测：私有实现细节（测行为，不测实现）

6.2 Sanitizer：让隐藏的内存错误立刻暴露

这是 C++ 开发中性价比最高的一个工具。它会拖慢程序 2～3 倍，但能抓住越界、悬垂指针、 内存泄漏、数据竞争、未定义行为——这些都是"平时没事、偶发崩溃"的元凶。 


# ============ GCC / Clang：编译时加开关即可 ============

# AddressSanitizer：越界、悬垂指针、double free、内存泄漏
g++ -std=c++17 -g -O1 -fsanitize=address,undefined -fno-omit-frame-pointer main.cpp -o app

# UndefinedBehaviorSanitizer：有符号溢出、除零、空指针解引用
g++ -std=c++17 -g -fsanitize=undefined main.cpp -o app

# ThreadSanitizer：数据竞争（注意不能与 ASan 同时用）
g++ -std=c++17 -g -fsanitize=thread main.cpp -o app -pthread

# 泄漏检测更详细（Linux）
ASAN_OPTIONS=detect_leaks=1 ./app

# ============ MSVC：在 Visual Studio 里开 ============
# 项目属性 → C/C++ → 常规 → 启用地址擦除器 AddressSanitizer) → 是 (/fsanitize=address)
# 或在 CMake 里：
#   if(MSVC)
#       add_compile_options(/fsanitize=address)
#       add_link_options(/fsanitize=address)
#   endif()
🎯
一个真实案例：为什么 ASan 值得用

越界写一个栈数组，可能覆盖掉相邻变量里的一个字节。程序不会立刻崩， 而是在几百行之后用一个莫名其妙的值算出错误结果。 靠 printf 找这种 bug 可能要花一整天；开了 ASan，它会在越界发生的那一行直接报出 stack-buffer-overflow 和完整的调用栈。

调试与发布必须用不同的构建类型

Debug：关优化（-O0 / /Od）、带调试符号（-g / /Zi）、断言有效。 断点靠它才能准确命中。
Release：开优化（-O2 / /O2）、NDEBUG 关闭断言、 性能测试必须用它，而且永远不要用 Release 调 bug——编译器会把代码重排得让你看不懂。
